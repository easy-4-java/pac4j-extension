package org.pac4j.core.ext.exception;


import org.pac4j.core.exception.TechnicalException;

/**
 * Exception thrown when the number of failed login attempts has reached the configured retry limit, requiring captcha verification.
 */
public class OverRetryRemindException extends TechnicalException {
    // ~ Constructors
    // ===================================================================================================

    /**
     * Constructs an <code>AuthenticationOverRetryRemindException</code> with the
     * specified message.
     *
     * @param msg the detail message
     */
    public OverRetryRemindException(String msg) {
        super(msg);
    }

    /**
     * Constructs an <code>AuthenticationOverRetryRemindException</code> with the
     * specified message and root cause.
     *
     * @param msg the detail message
     * @param t root cause
     */
    public OverRetryRemindException(String msg, Throwable t) {
        super(msg, t);
    }
}
